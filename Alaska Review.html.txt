    let currentCategory = 'ALL';
    const listEl = document.getElementById('list');
    const detailsEl = document.getElementById('details');
    const filterEl = document.getElementById('filter');
    const countEl = document.getElementById('count');

    function switchTab(tabId) {
      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
      document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
      event.target.classList.add('active');
      document.getElementById('tab-' + tabId).classList.add('active');
    }

    function setCategory(cat) {
      currentCategory = cat;
      document.querySelectorAll('.filter-chip').forEach(c => c.classList.remove('active'));
      event.target.classList.add('active');
      render(filterEl.value);
    }

    function render(term = '') {
      listEl.innerHTML = '';
      const matched = directory.filter(u => {
        const matchesTerm = u.name.toLowerCase().includes(term.toLowerCase()) || 
                            u.role.toLowerCase().includes(term.toLowerCase()) ||
                            u.email.toLowerCase().includes(term.toLowerCase());
        const matchesCat = currentCategory === 'ALL' || u.category === currentCategory;
        return matchesTerm && matchesCat;
      });
      countEl.textContent = matched.length;

      matched.forEach((user) => {
        const item = document.createElement('div');
        item.className = 'user-item';
        item.innerHTML = `
          <div>
            <div class="user-title">${user.name}</div>
            <div class="user-subtitle">${user.role}</div>
          </div>
          <span class="status-tag ${user.tag}">${user.status}</span>
        `;
        item.addEventListener('click', () => {
          document.querySelectorAll('.user-item').forEach(i => i.classList.remove('active'));
          item.classList.add('active');
          showUser(user);
        });
        listEl.appendChild(item);
      });
    }

    function showUser(user) {
      const pct = Math.round((user.mailbox_gb / user.quota_gb) * 100);
      detailsEl.innerHTML = `
        <div class="profile-heading">
          <h2>${user.name}</h2>
          <p>${user.email} &bull; <strong>${user.role}</strong></p>
        </div>

        <div class="metric-grid">
          <div class="metric-box">
            <div class="metric-name">Storage Utilized</div>
            <div class="metric-val">${user.mailbox_gb.toFixed(2)} GB</div>
            <div class="bar-bg">
              <div class="bar-fill" style="width:${Math.min(pct, 100)}%; background:${pct > 90 ? 'var(--danger)' : pct > 75 ? 'var(--warning)' : 'var(--alaska-green)'};"></div>
            </div>
          </div>
          <div class="metric-box">
            <div class="metric-name">Assigned Quota</div>
            <div class="metric-val">${user.quota_gb} GB</div>
            <div style="font-size:0.75rem; color:var(--text-muted); margin-top:0.45rem;">${pct}% total capacity used</div>
          </div>
          <div class="metric-box">
            <div class="metric-name">Online Archive State</div>
            <div class="metric-val">${user.archive === 'Yes' ? 'Active' : 'Disabled'}</div>
          </div>
          <div class="metric-box">
            <div class="metric-name">Last Mailbox Activity</div>
            <div class="metric-val" style="font-size:1.05rem;">${user.last_active}</div>
          </div>
        </div>

        <div>
          <div class="metric-name" style="margin-bottom:0.4rem;">Assigned License Subscriptions</div>
          <div class="badge-wrap">
            ${user.licenses.map(lic => `<span class="license-pill">${lic}</span>`).join('')}
          </div>
        </div>

        <div class="recommendation-panel ${user.urgent ? 'danger' : ''}">
          <h4>Actionable Optimization</h4>
          <p>${user.action}</p>
        </div>
      `;
    }

    filterEl.addEventListener('input', (e) => render(e.target.value));
    render();
    if (directory.length > 0) {
      document.querySelector('.user-item').classList.add('active');
      showUser(directory[0]);
    }
